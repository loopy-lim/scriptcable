cmake_minimum_required(VERSION 3.16)
project(scriptca_lynx_host LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT DEFINED SCRIPTCA_LIB_DIR)
  set(SCRIPTCA_LIB_DIR "${CMAKE_CURRENT_LIST_DIR}/../../../examples/calculator-core/.scriptca")
endif()
add_library(scriptca_lynx_host STATIC ScriptcaHost.cpp)
target_include_directories(scriptca_lynx_host PUBLIC ${CMAKE_CURRENT_LIST_DIR})
target_link_directories(scriptca_lynx_host PUBLIC ${SCRIPTCA_LIB_DIR})
target_link_libraries(scriptca_lynx_host PUBLIC calculator m pthread dl)
add_executable(scriptca_lynx_smoke smoke_main.cpp)
target_link_libraries(scriptca_lynx_smoke PRIVATE scriptca_lynx_host)
